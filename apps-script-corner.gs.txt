/**
 * BEM STDIIS | Samasta Corner: penerima absensi dan penilaian acara (Google Apps Script)
 * Tempel seluruh kode ini di Google Sheets: Ekstensi > Apps Script.
 * Panduan lengkap: PANDUAN-CORNER.txt
 *
 * Yang HARUS kamu ganti: KODE di bawah (kode rahasia tiap corner untuk membuka pembukuan).
 */
var SHEET_ID = '';            // OPSIONAL. Isi ID Spreadsheet bila skrip dibuat terpisah dari Sheet.
var SHEET_NAME = 'Absensi';
var KODE = {                  // id corner : kode rahasia. Tambah baris baru bila ada corner lain.
  irjen: 'GANTI-KODE-IRJEN'
};
var MAX_SALAH = 25;           // batas kode salah per 10 menit sebelum dikunci sementara

var KOLOM = ['Waktu', 'ID', 'Corner', 'Event', 'Nama', 'NIM', 'Bagian', 'Rating (JSON)', 'Rata-rata', 'Komentar', 'Kritik dan saran'];

function doPost(e) {
  try {
    var d = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (d.aksi === 'kirim') return kirim_(d);
    if (d.aksi === 'data') return data_(d);
    if (d.aksi === 'hapus') return hapus_(d);
    return reply_({ ok: false, error: 'aksi' });
  } catch (err) {
    console.error(err);
    return reply_({ ok: false, error: 'server' });
  }
}

// Buka URL Web App di browser: harus muncul "ok":true DAN "sheet":"terhubung"
function doGet() {
  var sh = null, msg = '';
  try { sh = sheet_(); } catch (err) { msg = String(err); }
  return reply_({
    ok: true,
    info: 'Endpoint Samasta Corner BEM STDIIS aktif',
    sheet: sh ? 'terhubung' : 'TIDAK terhubung ke Spreadsheet',
    kode: Object.keys(KODE).map(function (k) { return k + (KODE[k] && KODE[k].indexOf('GANTI') !== 0 ? ': sudah diganti' : ': BELUM DIGANTI'); }),
    detail: msg
  });
}

// Jalankan SEKALI secara manual (pilih "setup" lalu klik Jalankan) untuk memberi izin dan membuat tab "Absensi".
function setup() {
  var sh = sheet_();
  if (!sh) throw new Error('Tidak ada Spreadsheet. Pasang kode lewat Ekstensi > Apps Script dari dalam Sheet, atau isi SHEET_ID.');
  Logger.log('OK, tab "' + sh.getName() + '" siap di Spreadsheet: ' + sh.getParent().getUrl());
}

/* ---------- Peserta mengirim absensi dan penilaian ---------- */
function kirim_(d) {
  if (d.website) return reply_({ ok: true });                                   // jebakan bot
  var corner = clean_(d.corner, 30), ev = clean_(d.event, 40), nama = clean_(d.nama, 80),
      nim = clean_(d.nim, 20), bagian = clean_(d.bagian, 100),
      komentar = clean_(d.komentar, 600), kritik = clean_(d.kritik, 800);
  if (!/^[a-z0-9-]{1,30}$/.test(corner) || !/^[A-Za-z0-9_-]{1,40}$/.test(ev)) return reply_({ ok: false, error: 'data' });
  if (nama.length < 3 || !/^[A-Za-z0-9.\-\/]{4,20}$/.test(nim) || !bagian) return reply_({ ok: false, error: 'data' });

  var rating = {}, jumlah = 0, total = 0, r = d.rating;
  if (!r || typeof r !== 'object') return reply_({ ok: false, error: 'data' });
  Object.keys(r).slice(0, 12).forEach(function (k) {
    var v = Math.round(Number(r[k]));
    if (/^[A-Za-z0-9_]{1,24}$/.test(k) && v >= 1 && v <= 5) { rating[k] = v; jumlah++; total += v; }
  });
  if (!jumlah) return reply_({ ok: false, error: 'data' });

  var sh = sheet_();
  if (!sh) return reply_({ ok: false, error: 'nosheet' });
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var last = sh.getLastRow();
    if (last > 1) {                                                              // satu NIM hanya sekali per acara
      var v = sh.getRange(2, 4, last - 1, 3).getValues();                        // kolom D (Event) .. F (NIM)
      for (var i = 0; i < v.length; i++) {
        if (String(v[i][0]) === ev && String(v[i][2]).toLowerCase() === nim.toLowerCase()) return reply_({ ok: false, error: 'dup' });
      }
    }
    sh.appendRow([new Date(), Utilities.getUuid().slice(0, 8), corner, ev, nama, nim, bagian,
                  JSON.stringify(rating), Math.round(total / jumlah * 100) / 100, komentar, kritik]);
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }
  return reply_({ ok: true });
}

/* ---------- Irjen membaca pembukuan (butuh kode) ---------- */
function data_(d) {
  var corner = clean_(d.corner, 30);
  var g = periksaKode_(corner, d.kode);
  if (g) return reply_({ ok: false, error: g });
  var sh = sheet_();
  if (!sh) return reply_({ ok: false, error: 'nosheet' });
  var last = sh.getLastRow(), rows = [];
  if (last > 1) {
    sh.getRange(2, 1, last - 1, KOLOM.length).getValues().forEach(function (v) {
      if (String(v[2]) !== corner) return;
      var rt = {}; try { rt = JSON.parse(v[7]) || {}; } catch (x) {}
      rows.push({
        id: String(v[1]), t: v[0] instanceof Date ? v[0].toISOString() : String(v[0]), corner: String(v[2]), event: String(v[3]),
        nama: String(v[4]), nim: String(v[5]), bagian: String(v[6]), rating: rt, komentar: String(v[9] || ''), kritik: String(v[10] || '')
      });
    });
  }
  return reply_({ ok: true, rows: rows });
}

function hapus_(d) {
  var corner = clean_(d.corner, 30), id = clean_(d.id, 20);
  var g = periksaKode_(corner, d.kode);
  if (g) return reply_({ ok: false, error: g });
  var sh = sheet_();
  if (!sh || !id) return reply_({ ok: false, error: 'data' });
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var last = sh.getLastRow();
    if (last > 1) {
      var v = sh.getRange(2, 2, last - 1, 2).getValues();                         // kolom B (ID) dan C (Corner)
      for (var i = 0; i < v.length; i++) {
        if (String(v[i][0]) === id && String(v[i][1]) === corner) { sh.deleteRow(i + 2); SpreadsheetApp.flush(); return reply_({ ok: true }); }
      }
    }
  } finally {
    lock.releaseLock();
  }
  return reply_({ ok: false, error: 'notfound' });
}

/* ---------- Pemeriksaan kode + pembatasan percobaan ---------- */
function periksaKode_(corner, kode) {
  var cache = CacheService.getScriptCache(), kunci = 'salah_' + corner;
  var salah = Number(cache.get(kunci) || 0);
  if (salah >= MAX_SALAH) return 'tunggu';
  var benar = KODE[corner];
  var ok = !!benar && benar.indexOf('GANTI') !== 0 && sama_(String(kode || ''), String(benar));
  if (!ok) {
    cache.put(kunci, String(salah + 1), 600);
    Utilities.sleep(1200);                                                         // memperlambat tebakan
    return 'kode';
  }
  return '';
}
function sama_(a, b) {                                                             // perbandingan waktu-tetap
  var s = a.length === b.length ? 0 : 1;
  for (var i = 0; i < Math.max(a.length, b.length); i++) s |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0);
  return s === 0;
}

function clean_(v, max) {
  return String(v == null ? '' : v)
    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '')
    .trim()
    .slice(0, max);
}

function sheet_() {
  var ss = SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) return null;
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(KOLOM);
    sh.setFrozenRows(1);
    sh.getRange('1:1').setFontWeight('bold');
    // Semua kolom kecuali Waktu dan Rata-rata berformat teks agar isian tidak pernah dibaca sebagai rumus
    sh.getRange('B:H').setNumberFormat('@');
    sh.getRange('J:K').setNumberFormat('@');
    sh.getRange('A:A').setNumberFormat('dd/MM/yyyy HH:mm:ss');
    sh.getRange('J:K').setWrap(true);
    sh.setColumnWidth(1, 150); sh.setColumnWidth(5, 220); sh.setColumnWidth(7, 220);
    sh.setColumnWidth(8, 240); sh.setColumnWidth(10, 320); sh.setColumnWidth(11, 360);
  }
  return sh;
}

function reply_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
